🧠 2) scorer.py (Ralph’s brain logic)
def score_experiment(exp):
    score = 0

    # Attention
    if exp["impressions"] > 0:
        ctr = exp["clicks"] / exp["impressions"]
        score += ctr * 20

    # Engagement
    if exp["clicks"] > 0:
        reply_rate = exp["replies"] / exp["clicks"]
        score += reply_rate * 25

    # Intent
    if exp["replies"] > 0:
        dm_rate = exp["dms"] / exp["replies"]
        score += dm_rate * 25

    # Conversion
    if exp["dms"] > 0:
        lead_rate = exp["leads"] / exp["dms"]
        score += lead_rate * 20

    # Quality
    if exp["leads"] > 0:
        quality = exp["qualified_leads"] / exp["leads"]
        score += quality * 30

    return round(score, 2)


def classify(score):
    if score >= 70:
        return "SCALE"
    elif score >= 40:
        return "ITERATE"
    elif score >= 20:
        return "TEST"
    else:
        return "KILL"
⚙️ 3) app.py (simple dashboard server)
from flask import Flask, render_template
import json
from scorer import score_experiment, classify

app = Flask(__name__)

def load_data():
    with open("data.json") as f:
        return json.load(f)

@app.route("/")
def dashboard():
    data = load_data()
    experiments = data["experiments"]

    enriched = []
    for exp in experiments:
        score = score_experiment(exp)
        decision = classify(score)

        exp["score"] = score
        exp["decision"] = decision

        enriched.append(exp)

    # Sort by best performers
    enriched = sorted(enriched, key=lambda x: x["score"], reverse=True)

    return render_template("index.html", experiments=enriched)

app.run(host="0.0.0.0", port=3000)
