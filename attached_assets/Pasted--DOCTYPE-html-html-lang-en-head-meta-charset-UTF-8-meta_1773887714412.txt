async function generateCardsFromSite(url) {
  const domain = url.replace('https://','').replace('http://','').split('/')[0];

  const prompt = `You are a brutally honest business analyst and the IdeaRalph concept engine for the Vessel Protocol.

A user submitted this URL for scanning: ${url}

Generate a JSON response with this exact structure (no markdown, pure JSON):
{
  "blackCard": "A single Cards Against Humanity style black card sentence with ONE blank (use _______) that captures the core business challenge this type of site faces. Make it punchy, a little dark, honest.",
  "painCards": [
    "Pain point white card 1 (10-15 words max, specific, no fluff)",
    "Pain point white card 2",
    "Pain point white card 3",
    "Pain point white card 4"
  ],
  "agentCards": [
    "AI agent solution card 1 (how Vessel Protocol agents specifically solve one of the pain points)",
    "AI agent solution card 2",
    "AI agent solution card 3"
  ],
  "agentAnalysis": "A 3-sentence punchy analysis from ${servingAgent.name} (${servingAgent.niche} agent). Identify what this site is leaving on the table, what an AI agent would fix immediately, and end with a specific Vessel Protocol value prop. First person voice, direct, no hype words."
}

Make pain cards feel like someone finally said the quiet part out loud. Agent solution cards should feel like obvious fixes the site owner wishes they'd thought of.`;

  try {
    const res = await fetch("https://api.anthropic.com/v1/messages", {
      method:'POST',
      headers:{'Content-Type':'application/json'},
      body: JSON.stringify({
        model:'claude-sonnet-4-20250514',
        max_tokens:1000,
        messages:[{role:'user',content:prompt}]
      })
    });

    const data = await res.json();
    const raw = data.content?.map(b=>b.text||'').join('') || '';

    let parsed;
    try {
      const clean = raw.replace(/```json|```/g,'').trim();
      parsed = JSON.parse(clean);
    } catch(e) {
      parsed = getFallbackCards(domain);
    }

    renderCards(parsed);

  } catch(e) {
    renderCards(getFallbackCards(domain));
  }
}

function renderCards(data) {
  // Black card
  document.getElementById('bcText').innerHTML = data.blackCard.replace('_______','<span class="bc-blank">_______</span>');

  // White cards
  const grid = document.getElementById('whiteCardsGrid');
  grid.innerHTML = '';

  data.painCards.forEach((text, i) => {
    const card = createWhiteCard(text, 'pain', `PAIN POINT ${i+1}`);
    grid.appendChild(card);
  });

  data.agentCards.forEach((text, i) => {
    const card = createWhiteCard(text, 'agent', `AGENT FIX ${i+1}`);
    grid.appendChild(card);
  });

  // Agent analysis
  typeIn(document.getElementById('apAnalysis'), data.agentAnalysis);
}

function createWhiteCard(text, type, label) {
  const div = document.createElement('div');
  div.className = `white-card ${type}`;
  div.innerHTML = `<span class="card-type-tag">${label}</span>${text}`;
  div.onclick = () => div.classList.toggle('selected');
  return div;
}

function typeIn(el, text) {
  el.innerHTML = '';
  let i = 0;
  const iv = setInterval(()=>{
    if(i < text.length){ el.innerHTML += text[i]; i++; }
    else clearInterval(iv);
  }, 16);
}
